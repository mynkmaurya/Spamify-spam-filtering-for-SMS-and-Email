Spamify REST API Documentation

Integrate real-time spam scoring directly into your apps, email clients, or SMS gateways.

POST /api/predict 
# Python Example
import requests

url = "http://127.0.0.1:5000/api/predict"
payload = {
    "text": "Your account has been suspended! Verify here: http://bit.ly/bank-auth",
    "subject": "Urgent Security Alert",
    "mode": "email"
}

response = requests.post(url, json=payload)
data = response.json()
print("Classification:", data["classification"])
print("Spam Risk Score:", data["spam_score"], "%")
print("Trigger Words:", [t["word"] for t in data["trigger_words"]])
